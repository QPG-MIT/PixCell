The error `ValueError: min_bend_radius 0.322 < CrossSection.radius_min 5.0` occurs because the device footprint ($3 \mu m \times 5.5 \mu m$) requires bend radii ($\sim 1.25 \mu m$) that are much smaller than the default foundry checks in GDSFactory (usually $5 \mu m$).

To fix this, we must define a custom **CrossSection** with a relaxed `radius_min` constraint. I will also switch to `bend_circular` for the 180-degree turns to ensure the geometry strictly respects the radius dimensions for this specific footprint calculation.

Here is the corrected script:

```python
import gdsfactory as gf

@gf.cell
def pinched_racetrack_mzi():
    """
    Recreates the pinched racetrack device from the image within a 3x5.5um footprint.
    Uses a custom cross-section to allow small bend radii.
    """
    c = gf.Component()

    # --- 1. Define Custom Cross-Section ---
    # We need to override the default radius_min (usually 5um) to allow for
    # the tight bends required by the 3um height constraint.
    wg_width = 0.5
    xs = gf.cross_section.cross_section(
        width=wg_width,
        layer=(1, 0),
        radius=1.25,       # Desired radius for the main loops
        radius_min=0.01    # Relax constraint to allow tight S-bends
    )
    
    # --- 2. Calculate Dimensions ---
    # Constraints
    total_w = 5.5
    total_h = 3.0
    
    # Vertical Geometry
    # To fit height 3.0 with width 0.5, the centerline radius is (3.0 - 0.5)/2
    r_center = (total_h - wg_width) / 2  # 1.25 um
    
    # The straights are "pinched" inwards. 
    # Let's define the vertical offset for the inner straights.
    # Gap between inner straights ~0.6um
    inner_gap = 0.6
    y_straight_pos = (inner_gap + wg_width) / 2
    
    # Vertical distance the S-bend must cover (from R-center to Straight-center)
    dy_sbend = r_center - y_straight_pos
    
    # Horizontal Geometry
    # The 180 bend (circular) occupies exactly 'r_center' width.
    # Total Width = (Left Bend R) + (Left S-bend dx) + (Straight L) + (Right S-bend dx) + (Right Bend R)
    dx_sbend = 0.6  # Horizontal length of the S-transition
    l_straight = total_w - (2 * r_center) - (2 * dx_sbend) # 5.5 - 2.5 - 1.2 = 1.8

    # --- 3. Build Components ---

    # -- Top Arm --
    # Center Straight
    s_top = c << gf.components.straight(length=l_straight, cross_section=xs)
    s_top.move((0, y_straight_pos))
    s_top.movex(-l_straight/2) # Center at x=0

    # Right S-bend (upwards to match bend radius)
    sb_tr = c << gf.components.bend_s(size=(dx_sbend, dy_sbend), cross_section=xs)
    sb_tr.connect("o1", s_top.ports["o2"])

    # Left S-bend
    sb_tl = c << gf.components.bend_s(size=(dx_sbend, dy_sbend), cross_section=xs)
    sb_tl.mirror() 
    sb_tl.connect("o1", s_top.ports["o1"])

    # -- Bottom Arm --
    # Center Straight
    s_bot = c << gf.components.straight(length=l_straight, cross_section=xs)
    s_bot.move((0, -y_straight_pos))
    s_bot.movex(-l_straight/2)

    # Right S-bend (downwards to match bend radius)
    sb_br = c << gf.components.bend_s(size=(dx_sbend, -dy_sbend), cross_section=xs)
    sb_br.connect("o1", s_bot.ports["o2"])

    # Left S-bend
    sb_bl = c << gf.components.bend_s(size=(dx_sbend, -dy_sbend), cross_section=xs)
    sb_bl.mirror()
    sb_bl.connect("o1", s_bot.ports["o1"])

    # -- Loops (180 degree bends) --
    # Using bend_circular to ensure the width is exactly equal to radius
    
    # Right Loop
    bend_r = c << gf.components.bend_circular(radius=r_center, angle=180, cross_section=xs)
    # Connect the top of the bend (o2) to the top S-bend (o2)
    # Note: bend_circular ports are o1 (start) and o2 (end).
    # We need to rotate/mirror to align correctly. 
    # Standard bend starts horizontal.
    bend_r.connect("o2", sb_tr.ports["o2"])

    # Left Loop
    bend_l = c << gf.components.bend_circular(radius=r_center, angle=180, cross_section=xs)
    bend_l.connect("o1", sb_tl.ports["o2"])

    # --- 4. Heater Pads ---
    # Rectangles on the straight sections
    layer_heater = (2, 0)
    pad_w = l_straight / 2.2
    pad_h = wg_width * 0.9 
    pad_offset_x = l_straight / 3.8

    for ref_straight in [s_top, s_bot]:
        # Left pad
        p1 = c << gf.components.rectangle(size=(pad_w, pad_h), layer=layer_heater, centered=True)
        p1.move(ref_straight.center)
        p1.movex(-pad_offset_x)
        
        # Right pad
        p2 = c << gf.components.rectangle(size=(pad_w, pad_h), layer=layer_heater, centered=True)
        p2.move(ref_straight.center)
        p2.movex(pad_offset_x)

    return c

if __name__ == "__main__":
    device = pinched_racetrack_mzi()
    device.show()
    device.write_gds("pinched_racetrack_corrected.gds")
    
    # Verification
    print(f"Device name: {device.name}")
    print(f"Device bounding box: {device.bbox}")
```
