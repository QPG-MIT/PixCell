The error `ValueError: min_bend_radius 1.25 < CrossSection.radius_min 5.0` occurs because the default cross-section in GDSFactory enforces standard fabrication rules (often setting a minimum radius of 5um or 10um).

To fix this, we must explicitly define a `CrossSection` with a relaxed `radius_min` parameter that allows our tight 1.25um radius.

Here is the corrected script:

```python
import gdsfactory as gf

@gf.cell
def racetrack_structured():
    """
    Recreates the structured racetrack resonator.
    Constraint: 3um x 5.5um footprint.
    Fix: Defines a custom CrossSection to allow small bend radii.
    """
    c = gf.Component()
    
    # =======================================================================
    # 1. Design Parameters
    # =======================================================================
    layer = (1, 0)
    wg_width = 0.5
    
    # Geometry derivation from 3um x 5.5um footprint
    # Height = 3.0um. Radius = (Height - Width)/2 = 1.25um
    R = 1.25
    
    # Define a custom CrossSection that allows the small radius
    # We set radius_min=0.5 to satisfy the check (1.25 > 0.5)
    xs = gf.cross_section.cross_section(
        width=wg_width,
        layer=layer,
        radius=R,
        radius_min=0.5 
    )
    
    # Straight length derivation
    # Total Length = 5.5um.
    # L_straight = Total - 2 * (Radius + Width/2) = 5.5 - 3.0 = 2.5um
    L_straight = 2.5
    
    # Internal features
    wide_width = 0.9
    taper_len = 0.5
    block_len = L_straight - 2 * taper_len  # 1.5um
    
    # Slot/Hole calculations
    gap_size = 0.15
    hole_width = 0.5
    hole_len = (block_len - 3 * gap_size) / 2
    
    # =======================================================================
    # 2. Build the Structured Arm Sub-Component
    # =======================================================================
    arm = gf.Component("Structured_Arm")
    
    # --- Create the Central Slotted Block ---
    # 1. Solid Block
    solid = gf.components.rectangle(
        size=(block_len, wide_width), 
        layer=layer, 
        centered=True
    )
    
    # 2. Hole Geometry
    h_shape = gf.components.rectangle(
        size=(hole_len, hole_width), 
        layer=layer, 
        centered=True
    )
    
    # Position holes
    offset = (gap_size + hole_len) / 2
    h1 = h_shape.copy().move((-offset, 0))
    h2 = h_shape.copy().move((offset, 0))
    holes = gf.boolean(h1, h2, operation="or", layer=layer)
    
    # 3. Subtract Holes from Solid
    # Note: Boolean operations return a new component ref without ports
    slotted_block = gf.boolean(solid, holes, operation="not", layer=layer)
    arm << slotted_block
    
    # --- Add Tapers ---
    # Left Taper (Narrow -> Wide)
    t1 = gf.components.taper(
        length=taper_len, width1=wg_width, width2=wide_width, layer=layer
    )
    ref_t1 = arm << t1
    # Move taper end to left edge of block
    ref_t1.move(destination=(-block_len/2, 0), origin=ref_t1.ports["o2"])
    
    # Right Taper (Wide -> Narrow)
    t2 = gf.components.taper(
        length=taper_len, width1=wide_width, width2=wg_width, layer=layer
    )
    ref_t2 = arm << t2
    # Move taper start to right edge of block
    ref_t2.move(destination=(block_len/2, 0), origin=ref_t2.ports["o1"])
    
    # Expose ports
    arm.add_port("o1", port=ref_t1.ports["o1"])
    arm.add_port("o2", port=ref_t2.ports["o2"])
    
    # =======================================================================
    # 3. Assembly
    # =======================================================================
    
    # Instantiate components
    # Pass the custom cross_section 'xs' to the bend to avoid radius errors
    bend_geo = gf.components.bend_circular(radius=R, angle=180, cross_section=xs)
    
    b_left = c << bend_geo
    b_right = c << bend_geo
    top_arm = c << arm
    bot_arm = c << arm
    
    # Connect
    # 1. Anchor b_left
    b_left.move((0, 0))
    
    # 2. Connect Top Arm to top of b_left (port o2)
    top_arm.connect("o1", b_left.ports["o2"])
    
    # 3. Connect b_right to end of Top Arm
    # Connecting o2 to o2 flips the bend correctly for the right side
    b_right.connect("o2", top_arm.ports["o2"])
    
    # 4. Connect Bottom Arm to bottom of b_right (port o1)
    # Connecting o2 to o1 aligns it running right-to-left
    bot_arm.connect("o2", b_right.ports["o1"])
    
    return c

if __name__ == "__main__":
    # Create device
    device = racetrack_structured()
    
    # Output
    device.show()
    device.write_gds("racetrack_structured.gds")
    
    # Verify Footprint
    bbox = device.bbox
    w = bbox[1][0] - bbox[0][0]
    h = bbox[1][1] - bbox[0][1]
    print(f"Device Footprint: {w:.2f} x {h:.2f} um")
```
