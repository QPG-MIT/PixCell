The `ValueError` arises because the automatically generated S-bends in the previous script had a curvature radius smaller than the default strict minimum (10µm) or the specific radius I set (1.325µm). To fix this, we must explicitly allow tight bends in the `CrossSection` by setting `radius_min` to a very small value.

Additionally, I have recalculated the dimensions to strictly match the **5.5µm width constraint** (shortening the straights slightly) and ensured the loop geometry closes perfectly.

```python
import gdsfactory as gf

@gf.cell
def racetrack_mzi_modulator():
    """
    Recreates the racetrack resonator device with internal phase shifters.
    Adjusted for [3, 5.5] um footprint and fixes bend radius validation errors.
    """
    c = gf.Component()

    # --- 1. Dimensions & Constraints ---
    # Footprint: 3.0 um Height, 5.5 um Width
    total_height = 3.0
    target_width = 5.5
    wg_width = 0.35
    
    # Radius calculation (center-to-center vertical)
    # Height = 2*R + wg_width  =>  R = (3.0 - 0.35)/2
    radius = (total_height - wg_width) / 2  # 1.325 um
    
    # Width calculation & Length fitting
    # Width = 2*(R + wg_width/2) + Length_straight_section
    # 5.5 = 2*(1.325 + 0.175) + L_straight
    # 5.5 = 3.0 + L_straight  =>  L_straight = 2.5 um
    # Straight section = 2 * sbend + heater_straight
    # Let's pick:
    sbend_length = 0.7
    heater_length = 1.1 
    # Check: 2*0.7 + 1.1 = 2.5. Perfect.

    inner_gap = 0.4
    
    # --- 2. Define Cross Section with Relaxed Constraints ---
    # radius_min must be small to allow the tight S-bends
    xs = gf.cross_section.cross_section(
        width=wg_width,
        layer=(1, 0),
        radius=radius,      # Default radius for auto-bends
        radius_min=0.01     # CRITICAL FIX: Allow tight curvature for S-bends
    )

    # --- 3. Construct Left Loop (Cap) ---
    # Rotated 180deg: Starts at Top (o1), Ends at Bottom (o2)
    lbend = c << gf.components.bend_euler(
        radius=radius, 
        angle=180, 
        p=0.5, 
        cross_section=xs
    )
    lbend.rotate(180)
    # Move to center vertically around y=0
    # Original (0,0)->(0,2R) becomes (0,0)->(0,-2R) after rotate
    # Shift up by R to make it +R to -R
    lbend.movey(radius)

    # --- 4. Top Arm (Left to Right) ---
    # Path: Connects to lbend.o1 (Top, facing West - we connect o1 of arm here)
    
    # Calculate S-bend vertical drop
    # Start Y = radius. End Y = (inner_gap + wg_width)/2
    y_top_inner = (inner_gap + wg_width) / 2
    dy_top = y_top_inner - radius 

    # S-bend down
    s_top_1 = c << gf.components.bend_s(
        size=(sbend_length, dy_top), 
        cross_section=xs
    )
    s_top_1.connect("o1", lbend.ports["o1"])

    # Straight Heater
    straight_top = c << gf.components.straight(
        length=heater_length, 
        cross_section=xs
    )
    straight_top.connect("o1", s_top_1.ports["o2"])

    # S-bend up
    s_top_2 = c << gf.components.bend_s(
        size=(sbend_length, -dy_top), 
        cross_section=xs
    )
    s_top_2.connect("o1", straight_top.ports["o2"])

    # --- 5. Right Loop (Cap) ---
    rbend = c << gf.components.bend_euler(
        radius=radius, 
        angle=180, 
        p=0.5, 
        cross_section=xs
    )
    rbend.connect("o1", s_top_2.ports["o2"])

    # --- 6. Bottom Arm (Right to Left) ---
    # Path: Connects to rbend.o2 (Bottom, facing West)
    
    # Calculate S-bend vertical rise
    # Start Y = -radius. End Y = -(inner_gap + wg_width)/2
    y_bot_inner = -(inner_gap + wg_width) / 2
    dy_bot = y_bot_inner - (-radius)

    # S-bend up
    s_bot_1 = c << gf.components.bend_s(
        size=(sbend_length, dy_bot), 
        cross_section=xs
    )
    s_bot_1.connect("o1", rbend.ports["o2"])

    # Straight Heater
    straight_bot = c << gf.components.straight(
        length=heater_length, 
        cross_section=xs
    )
    straight_bot.connect("o1", s_bot_1.ports["o2"])

    # S-bend down
    s_bot_2 = c << gf.components.bend_s(
        size=(sbend_length, -dy_bot), 
        cross_section=xs
    )
    s_bot_2.connect("o1", straight_bot.ports["o2"])
    
    # Note: s_bot_2.o2 should align with lbend.o2

    # --- 7. Add Heater Pads ---
    # Visually recreate the pads as rectangles on the straight sections
    pad_width = 0.55
    pad_len_rect = (heater_length / 2) - 0.1
    
    pad = gf.components.rectangle(
        size=(pad_len_rect, pad_width), 
        layer=(1, 0), 
        centered=True
    )

    def add_pads(ref):
        # Left sub-pad
        p1 = c << pad
        p1.move(ref.center)
        p1.movex(-(pad_len_rect/2 + 0.05))
        # Right sub-pad
        p2 = c << pad
        p2.move(ref.center)
        p2.movex(+(pad_len_rect/2 + 0.05))

    add_pads(straight_top)
    add_pads(straight_bot)

    return c

if __name__ == "__main__":
    device = racetrack_mzi_modulator()
    device.show()
    device.write_gds("racetrack_mzi.gds")
```
