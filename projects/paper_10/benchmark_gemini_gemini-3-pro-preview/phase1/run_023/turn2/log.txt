import gdsfactory as gf
from gdsfactory.components import rectangle, taper, straight, bend_euler

@gf.cell
def device_racetrack_mzm_v2():
    """
    Recreates the racetrack resonator with corrected topology.
    - Fixes the open-loop 'S-shape' by daisy-chaining connections: Top -> Bend -> Bottom -> Bend.
    - Uses explicit radius definitions to meet the tight 3um x 5.5um footprint.
    """
    c = gf.Component()

    # --- 1. Geometric Parameters ---
    # Footprint constraints: 3.0um height, 5.5um width.
    
    # Height analysis: 3.0um total height.
    # Structure height = 2 * Radius + Bend_Width.
    # 3.0 = 2*R + 0.7  =>  2R = 2.3  =>  Radius = 1.15um.
    bend_radius = 1.15
    bend_width = 0.7
    
    # Length analysis: 5.5um total width.
    # Structure width = Straight_Length + 2 * Bend_X_Size.
    # For an Euler 180 bend with R=1.15, the X-size is approx 1.3~1.5um.
    # We choose a straight length to fit the remaining space (~2.5 - 2.8um).
    straight_section_length = 2.5
    
    # Inner arm parameters
    arm_width = 0.35
    taper_length = 0.4
    
    # Pad parameters
    pad_width = 0.8
    pad_height = 0.6
    pad_spacing = 0.4

    # --- 2. Custom Cross Sections ---
    # 'radius_min' is set low to permit the tight 1.15um bend without error.
    xs_bend = gf.cross_section.cross_section(
        width=bend_width, 
        layer=(1, 0),
        radius=bend_radius,
        radius_min=0.5 
    )
    
    xs_arm = gf.cross_section.cross_section(
        width=arm_width, 
        layer=(1, 0)
    )

    # --- 3. Define the Straight Arm Component ---
    @gf.cell
    def active_arm():
        """Creates the straight section with tapers and heater pads."""
        cmp = gf.Component()
        
        # 1. Input Taper (Bend Width -> Arm Width)
        t_in = cmp << taper(length=taper_length, width1=bend_width, width2=arm_width, layer=(1,0))
        
        # 2. Straight Waveguide
        wg = cmp << straight(length=straight_section_length, cross_section=xs_arm)
        wg.connect("o1", t_in.ports["o2"])
        
        # 3. Output Taper (Arm Width -> Bend Width)
        t_out = cmp << taper(length=taper_length, width1=arm_width, width2=bend_width, layer=(1,0))
        t_out.connect("o1", wg.ports["o2"])
        
        # 4. Rectangular Pads
        center_x = taper_length + straight_section_length / 2
        
        # Pad 1
        p1 = cmp << rectangle(size=(pad_width, pad_height), layer=(1,0), centered=True)
        p1.move((center_x - pad_width/2 - pad_spacing/2, 0))
        
        # Pad 2
        p2 = cmp << rectangle(size=(pad_width, pad_height), layer=(1,0), centered=True)
        p2.move((center_x + pad_width/2 + pad_spacing/2, 0))
        
        # Expose ports
        cmp.add_port("o1", port=t_in.ports["o1"])
        cmp.add_port("o2", port=t_out.ports["o2"])
        return cmp

    # --- 4. Assembly (Daisy Chain Method) ---
    # We connect components in a sequence (Top -> Right -> Bottom -> Left) 
    # to guarantee a closed loop.
    
    # 1. Place Top Arm
    top_arm = c << active_arm()
    
    # 2. Connect Right Bend
    # Connect bend input (o1) to top arm output (o2)
    bend_r = c << bend_euler(radius=bend_radius, angle=180, cross_section=xs_bend)
    bend_r.connect("o1", top_arm.ports["o2"])
    
    # 3. Connect Bottom Arm
    # Connect bottom arm (o2) to bend output (o2)
    # Note: Connecting o2-to-o2 effectively aligns the component "backwards",
    # so it runs from right to left, closing the loop.
    bot_arm = c << active_arm()
    bot_arm.connect("o2", bend_r.ports["o2"])
    
    # Optional: Mirror bottom arm geometry if vertical symmetry is required for pads
    # (Since pads are vertically centered in the arm, this is visually identical, but good practice)
    # bot_arm.mirror_y() 
    
    # 4. Connect Left Bend
    # Connect bend input (o1) to bottom arm input (o1)
    bend_l = c << bend_euler(radius=bend_radius, angle=180, cross_section=xs_bend)
    bend_l.connect("o1", bot_arm.ports["o1"])
    
    # The loop is now topologically closed. 
    # bend_l.ports["o2"] should inherently align with top_arm.ports["o1"].

    return c

if __name__ == "__main__":
    device = device_racetrack_mzm_v2()
    device.show()
    device.write_gds("racetrack_mzm_v2.gds")
