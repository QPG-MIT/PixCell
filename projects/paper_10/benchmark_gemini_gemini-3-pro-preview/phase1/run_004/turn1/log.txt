To fix this, we must explicitly define a **Custom Cross-Section** with `radius_min=0` (or small enough, e.g., 0.1) and pass this specific cross-section to the bend components. We cannot rely on the default "strip" or "generic" cross-sections.

import gdsfactory as gf

@gf.cell
def racetrack_mzi_fixed():
    """
    Recreates the racetrack device with internal heater pads.
    Constraint: 3um height x 5.5um width.
    Fixes: Explicitly sets radius_min to 0 to allow tight bends (R=1.2um).
    """
    c = gf.Component()

    # --- 1. Dimensions & Constraints ---
    h_total = 3.0
    w_total = 5.5
    
    width_ring = 0.6
    width_pad = 0.65
    width_thin = 0.25
    
    layer = (1, 0)

    # --- 2. Define Custom Cross-Section ---
    # CRITICAL FIX: We create a raw cross_section with radius_min=0.
    # This bypasses the default PDK check (usually 10um or 5um).
    xs_ring = gf.cross_section.cross_section(
        width=width_ring,
        layer=layer,
        radius_min=0,      # Disable minimum radius check
        radius=1.2         # Set default radius to our target (optional but good practice)
    )

    # --- 3. Build Outer Racetrack ---
    # Math: Outer Radius = 1.5. Ring Width = 0.6.
    # Center Radius = 1.5 - (0.6/2) = 1.2um.
    r_center = h_total / 2.0 - width_ring / 2.0  # 1.2 um
    l_straight = w_total - h_total               # 2.5 um

    # Create primitives using the CUSTOM cross-section
    bend = gf.components.bend_circular(
        radius=r_center, 
        angle=180, 
        cross_section=xs_ring
    )
    straight = gf.components.straight(
        length=l_straight, 
        cross_section=xs_ring
    )
    
    # Place Racetrack Parts
    # Top Straight
    s_top = c << straight
    s_top.movey(r_center)
    
    # Bottom Straight
    s_bot = c << straight
    s_bot.movey(-r_center)
    
    # Left Bend
    b_left = c << bend
    b_left.connect("o1", s_top.ports["o1"])
    
    # Right Bend
    b_right = c << bend
    b_right.connect("o2", s_top.ports["o2"])

    # --- 4. Build Internal Arm ---
    # Helper function to build one arm structure
    def create_arm():
        arm_c = gf.Component()
        
        # Dimensions for arm segments
        l_pad = 0.7
        l_taper = 0.2
        l_gap = 0.2
        l_connect = 0.6 # Length to reach the bend wall
        
        # Primitives
        s_connect = gf.components.straight(length=l_connect, width=width_thin, layer=layer)
        taper_up = gf.components.taper(length=l_taper, width1=width_thin, width2=width_pad, layer=layer)
        pad = gf.components.straight(length=l_pad, width=width_pad, layer=layer)
        taper_down = gf.components.taper(length=l_taper, width1=width_pad, width2=width_thin, layer=layer)
        s_gap = gf.components.straight(length=l_gap, width=width_thin, layer=layer)

        # Assemble: Connect -> Taper -> Pad -> Taper -> Gap -> Taper -> Pad -> Taper -> Connect
        # Left side
        c1 = arm_c << s_connect
        t1 = arm_c << taper_up
        t1.connect("o1", c1.ports["o2"])
        p1 = arm_c << pad
        p1.connect("o1", t1.ports["o2"])
        t2 = arm_c << taper_down
        t2.connect("o1", p1.ports["o2"])
        
        # Center gap
        g = arm_c << s_gap
        g.connect("o1", t2.ports["o2"])
        
        # Right side
        t3 = arm_c << taper_up
        t3.connect("o1", g.ports["o2"])
        p2 = arm_c << pad
        p2.connect("o1", t3.ports["o2"])
        t4 = arm_c << taper_down
        t4.connect("o1", p2.ports["o2"])
        c2 = arm_c << s_connect
        c2.connect("o1", t4.ports["o2"])
        
        return arm_c, c2.ports["o2"].center[0] # Return component and total length

    # Create the arm component
    arm_comp, arm_length = create_arm()

    # --- 5. Place Arms ---
    y_offset = 0.45
    
    # Center calculation:
    # The ring straight section is centered at x = l_straight/2 = 1.25
    # The arm center is at arm_length/2
    # We want to align the centers.
    x_pos = (l_straight / 2.0) - (arm_length / 2.0)

    # Top Arm
    ref_top = c << arm_comp
    ref_top.move((x_pos, y_offset))
    
    # Bottom Arm
    ref_bot = c << arm_comp
    ref_bot.move((x_pos, -y_offset))

    # --- 6. Boolean Merge ---
    # Merge everything into a single polygon to look like the image (removing overlaps)
    merged = gf.boolean(c, c, operation="or", layer=layer)
    
    # Final cleanup
    final_c = gf.Component("Racetrack_Final")
    ref_final = final_c << merged
    
    # Expose ports (mapped from original position)
    final_c.add_port("o1", center=(0, -r_center), width=width_ring, orientation=180, layer=layer)
    final_c.add_port("o2", center=(l_straight, -r_center), width=width_ring, orientation=0, layer=layer)

    return final_c

if __name__ == "__main__":
    # Create and show device
    device = racetrack_mzi_fixed()
    device.show()
    
    # Save
    device.write_gds("racetrack_mzi_fixed.gds")
